# 成都农商行电子对账单识别实施计划

> **执行要求：** 使用 executing-plans 在当前任务内逐项实施。每项使用复选框跟踪，并严格执行测试先行。

**目标：** 为现有 PDF 表格提取工具新增成都农商行固定版式识别，并把3份PDF中的17笔流水整理为可核对的Excel。

**架构：** 新增一个独立的成都农商行提取器，使用 pdfplumber 的表格结果读取十个固定栏目，并复用现有模型、验证器、批处理服务和Excel导出器。提取器先把倒序对账单恢复为日期升序和同日余额连续顺序，再交给统一验证器核对笔数、借贷合计和余额。

**技术栈：** Python 3、pdfplumber、Decimal、pytest、现有 openpyxl 导出器。

## 全局约束

- 仅支持 Windows 路径，不增加跨平台适配。
- 不增加第三方依赖，不调用大模型。
- 不修改民生、华夏和通用表格的现有识别行为。
- 原PDF只读，处理前后逐文件核对SHA256。
- Excel输出到农商行目录下的“提取结果”，包含“流水明细”和“提取汇总”。
- 无交易时刻的记录保留真实日期，时间为00:00:00，不编造具体时刻。
- 6月文件期间采用真实交易范围2025-04-02至2025-06-27，并在提取方式中保留原表月份。

- [ ] **步骤3：实现最小提取器**

实现以下固定规则，生产文件使用相同字段映射和排序逻辑：

```python
HEADERS = ["序号", "子序号", "交易日期", "摘要", "币种", "借方金额", "贷方金额", "账户余额", "对方账号", "对方户名"]


def _clean(value: object) -> str:
    return "".join(str(value or "").split())


def _decimal(value: object) -> Decimal:
    return Decimal(_clean(value).replace(",", "") or "0")


def is_chengdu_rural_statement(text: str) -> bool:
    compact = "".join((text or "").split())
    return all(label in compact for label in ("银行电子对账单", "账号", "交易日期", "借方金额", "贷方金额", "账户余额"))


def parse_chengdu_rural_pages(page_texts: list[str], page_tables: list[list[list]], source_path: Path) -> StatementExtraction:
    full_text = "\n".join(page_texts)
    compact = "".join(full_text.split())
    account = re.search(r"账号[：:](\d+)", compact).group(1)
    month = re.search(r"月份[：:](\d{4}年\d{2}月)", compact).group(1)
    declared_debit = _decimal(re.search(r"借款方总额[：:]([0-9,.]+)", compact).group(1))
    declared_credit = _decimal(re.search(r"贷款方总额[：:]([0-9,.]+)", compact).group(1))
    transactions = []
    sequences = []
    currencies = set()
    for page_number, tables in enumerate(page_tables, start=1):
        for table in tables:
            if not table or [_clean(value) for value in table[0]] != HEADERS:
                continue
            for row in table[1:]:
                sequence = int(_clean(row[0]))
                subsequence = _clean(row[1])
                occurred_at = datetime.strptime(_clean(row[2]), "%Y%m%d")
                counterparty_name = _clean(row[9])
                counterparty_account = _clean(row[8])
                counterparty = "/".join(value for value in (counterparty_name, counterparty_account) if value)
                currencies.add(_clean(row[4]))
                sequences.append(sequence)
                transactions.append(Transaction(
                    occurred_at=occurred_at,
                    summary=_clean(row[3]),
                    voucher_type="",
                    voucher_number=str(sequence),
                    debit=_decimal(row[5]),
                    credit=_decimal(row[6]),
                    balance=_decimal(row[7]),
                    serial_number=subsequence,
                    counterparty=counterparty,
                    counterparty_bank="",
                    bank_name="成都农村商业银行股份有限公司",
                    account_number=account,
                    period_start=None,
                    period_end=None,
                    source_file=source_path.name,
                    source_page=page_number,
                    extraction_method="",
                ))
    transactions.sort(key=lambda item: (item.occurred_at, -int(item.voucher_number)))
    period_start = min(item.occurred_at.date() for item in transactions)
    period_end = max(item.occurred_at.date() for item in transactions)
    method = f"成都农商行固定表格（原表月份：{month}；币种：{'、'.join(sorted(currencies))}）"
    for item in transactions:
        item.period_start = period_start
        item.period_end = period_end
        item.extraction_method = method
    declared_count = max(sequences, default=0)
    errors = [] if sorted(sequences) == list(range(1, declared_count + 1)) else [f"{source_path.name}：交易序号不连续"]
    return StatementExtraction(
        source_path=source_path,
        bank_name="成都农村商业银行股份有限公司",
        account_number=account,
        period_start=period_start,
        period_end=period_end,
        page_count=len(page_texts),
        extraction_method=method,
        transactions=transactions,
        declared_count=declared_count,
        declared_debit=declared_debit,
        declared_credit=declared_credit,
        errors=errors,
    )


def extract_chengdu_rural(pdf_path: Path) -> StatementExtraction | None:
    source_path = Path(pdf_path)
    with pdfplumber.open(source_path) as pdf:
        page_texts = [page.extract_text() or "" for page in pdf.pages]
        if not page_texts or not is_chengdu_rural_statement(page_texts[0]):
            return None
        page_tables = [page.extract_tables() or [] for page in pdf.pages]
    return parse_chengdu_rural_pages(page_texts, page_tables, source_path)
```
